fn filter_sum(f: Fn, n: Int) -> Int
    let mut sum = 0
    let mut i = 0
    while i < n
        if f(i)
            sum = sum + i
        end
        i = i + 1
    end
    sum
end

fn main() -> Unit
    let is_even = fn(n: Int) -> Bool
        if n % 2 == 0
            True
        else
            False
        end
    end
    println(filter_sum(is_even, 10))
end

fn safe_divide(a: Int, b: Int) -> Result<Int, String>
    if b == 0
        Err("div by zero")
    else
        Ok(a / b)
    end
end

fn main() -> Unit
    match safe_divide(10, 2)
        Ok(n) => println(n)
        Err(e) => println("Error: " + e)
    end
    match safe_divide(10, 0)
        Ok(n) => println(n)
        Err(e) => println("Error: " + e)
    end
end

fn first_char(s: String) -> Option<String>
    if len(s) == 0
        None
    else
        Some(s)
    end
end

        None => println("empty")
    end
    match first_char("")
        Some(s) => println("got: " + s)
        None => println("empty")
    end
end

enum LogLevel = Debug | Info | Warn | Error

fn log_prefix(level: LogLevel) -> String
    match level
        Debug => "[DBG]"
        Info => "[INF]"
        Warn => "[WRN]"
        Error => "[ERR]"
    end
end

fn log_message(level: LogLevel, msg: String) -> String
    log_prefix(level) + " " + msg
end

fn main() -> Unit
    println(log_message(Info, "started"))
    println(log_message(Error, "crashed"))
    println(log_message(Debug, "x=42"))
end

    if x < 0
        -x
    else
        x
    end
end

fn main() -> Unit
    println(negate(5))
    println(negate(-3))

    println(abs_val(4))
end

=== 008.lom ===
fn max3(a: Int, b: Int, c: Int) -> Int
    if a > b
        if a > c
            a
        else
            c
        end
    else
        if b > c
            b
        else
            c
        end
    end
end

fn main() -> Unit
    println(max3(3, 7, 5))
    println(max3(10, 2, 8))
    println(max3(1, 1, 1))
end

fn circle_area(r: Float) -> Float
    3.14159 * r * r
end

fn circle_perimeter(r: Float) -> Float
    2.0 * 3.14159 * r
end

fn main() -> Unit
    println(circle_area(2.0))
    println(circle_perimeter(2.0))

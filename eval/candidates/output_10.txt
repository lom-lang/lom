fn safe_divide(a: Int, b: Int) -> Result<Int, String>
    if b == 0
        Err("div by zero")
    else
        Ok(a / b)
    end
end

fn main() -> Unit
    match safe_divide(10, 2)
        Ok(n) => println(n)
        Err(e) => println("Error: " + e)
    end

    match safe_divide(10, 0)
        Ok(n) => println(n)
        Err(e) => println("Error: " + e)
    end
end

fn fib(n: Int) -> Int
    if n == 0
        0
    elif n == 1
        1
    else
        fib(n - 1) + fib(n - 2)
    end
end

fn main() -> Unit
    let mut i = 0
    while i < 5
        println(fib(i))
        i = i + 1
    end
end
